import os import duckdb import numpy as np import pandas as pd print("=" * 80) print("EXPANZE VÝBĚRU NA CELOU POPULACI S VÁŽENÝM PRŮMĚREM (RD+BD)") print("(fyzická expanze: celé budovy + umělé zbytky, bez nahrazování NaN vah)") print("=" * 80) # -------------------------------------------------- # 1. NAČTENÍ REÁLNÉ STRUKTURY POPULACE Z DUCKDB # -------------------------------------------------- db_path = r"C:\Users\16880\OneDrive - Ministerstvo financí ČR\DUCKDB - DUCKDB\buildings.duckdb" print(f"Připojuji se k databázi DuckDB: {db_path}...") con = duckdb.connect(db_path) print("Načítám reálné počty budov z populace...") df_populace_counts = con.execute(""" SELECT ENEX_kategorie, COUNT(*) as Pocet_v_Populaci_Real FROM kategorie GROUP BY ENEX_kategorie """).df() con.close() celkem_populace_db_vse = df_populace_counts["Pocet_v_Populaci_Real"].sum() print(f"Úspěšně načteno. Celková populace v DB (VŠECHNY kategorie): {celkem_populace_db_vse:,} budov.") # -------------------------------------------------- # 2. NAČTENÍ DETAILNÍHO VÝBĚROVÉHO SOUBORU S ÚSPORAMI # -------------------------------------------------- vyber_path = "ENEX_s_usporami_opatreni_rezidence.csv" print(f"\nNačítám podrobná výběrová data z {vyber_path}...") if not os.path.exists(vyber_path): raise FileNotFoundError( f"Soubor {vyber_path} nebyl nalezen! Ujistěte se, že máte spuštěný " f"předchozí krok výpočtu úspor." ) df_vyber = pd.read_csv(vyber_path, sep=";", decimal=",", low_memory=False) # Striktní extrakce/čištění třídy A-G pro distribuční klíč df_vyber["trida_U"] = df_vyber["souc_pro_tepla_zat"].astype(str).str.strip().str.upper() platne_tridy = ["A", "B", "C", "D", "E", "F", "G"] df_vyber["trida_U"] = np.where(df_vyber["trida_U"].isin(platne_tridy), df_vyber["trida_U"], "Neznámo") # Číselné ošetření klíčových energetických sloupců for col in ["celk_dod_ener", "celkprim", "PlochaEvzB", "combined_weight"]: if col in df_vyber.columns: df_vyber[col] = pd.to_numeric( df_vyber[col].astype(str).str.replace(",", ".", regex=False), errors="coerce" ) elif col == "combined_weight": print("Upozornění: Sloupec 'combined_weight' nenalezen, nastavuji 1.0") df_vyber[col] = 1.0 if "skupina_rdbd" in df_vyber.columns: print(f"Načteno: {len(df_vyber):,} řádků " f"(RD: {(df_vyber['skupina_rdbd'] == 'RD').sum():,}, " f"BD: {(df_vyber['skupina_rdbd'] == 'BD').sum():,}).") # -------------------------------------------------- # 2a. NORMALIZACE VAH PODLE POPULACE 2020 (R KÓD) # -------------------------------------------------- print("\n--- Počítám normalizovanou váhu (combined_weight_norm) ---") # ZADEJTE CÍLOVÉ PODLAHOVÉ PLOCHY ROKU 2020 (pop_rd a pop_bd): pop_rd = 12345678.9 # Doplňte přesnou hodnotu pop_rd pro RD (m²) pop_bd = 9876543.2 # Doplňte přesnou hodnotu pop_bd pro BD (m²) if "skupina_rdbd" in df_vyber.columns: mask_rd = (df_vyber["skupina_rdbd"] == "RD") & df_vyber["PlochaEvzB"].notna() & df_vyber["combined_weight"].notna() mask_bd = (df_vyber["skupina_rdbd"] == "BD") & df_vyber["PlochaEvzB"].notna() & df_vyber["combined_weight"].notna() mass_rd = (df_vyber.loc[mask_rd, "PlochaEvzB"] * df_vyber.loc[mask_rd, "combined_weight"]).sum() mass_bd = (df_vyber.loc[mask_bd, "PlochaEvzB"] * df_vyber.loc[mask_bd, "combined_weight"]).sum() scale_rd = pop_rd / mass_rd if mass_rd > 0 else np.nan scale_bd = pop_bd / mass_bd if mass_bd > 0 else np.nan df_vyber["combined_weight_norm"] = np.nan df_vyber.loc[df_vyber["skupina_rdbd"] == "RD", "combined_weight_norm"] = df_vyber.loc[df_vyber["skupina_rdbd"] == "RD", "combined_weight"] * scale_rd df_vyber.loc[df_vyber["skupina_rdbd"] == "BD", "combined_weight_norm"] = df_vyber.loc[df_vyber["skupina_rdbd"] == "BD", "combined_weight"] * scale_bd else: df_vyber["combined_weight_norm"] = df_vyber["combined_weight"] # -------------------------------------------------- # 1b. OMEZENÍ POPULAČNÍHO ZÁKLADU NA KATEGORIE VE VÝBĚRU # -------------------------------------------------- kategorie_ve_vyberu = set(df_vyber["ENEX_kategorie"].unique()) maska_relevantni_kat = df_populace_counts["ENEX_kategorie"].isin(kategorie_ve_vyberu) celkem_populace_db = df_populace_counts.loc[maska_relevantni_kat, "Pocet_v_Populaci_Real"].sum() kategorie_db_bez_vyberu = sorted( set(df_populace_counts.loc[~maska_relevantni_kat, "ENEX_kategorie"].unique()) ) print(f"\nPopulační základ omezen na kategorie přítomné ve výběru " f"({sorted(kategorie_ve_vyberu)}): {celkem_populace_db:,} budov " f"(z {celkem_populace_db_vse:,} v celé DB).") if kategorie_db_bez_vyberu: print(f" Kategorie v DB MIMO výběr (ignorovány): {kategorie_db_bez_vyberu}") # -------------------------------------------------- # 2b. DIAGNOSTIKA CHYBĚJÍCÍCH VAH # -------------------------------------------------- pocet_chybejici_vaha = df_vyber["combined_weight"].isna().sum() print(f"\nPočet řádků s chybějící 'combined_weight' (NaN): {pocet_chybejici_vaha:,}") # -------------------------------------------------- # 3. VÝPOČET DISTRIBUČNÍCH VAH (mezi podskupinami) # -------------------------------------------------- print("\nPočítám vážené podíly kombinací [Kategorie + Vypočtená kategorie + Třída]...") GRP = ["ENEX_kategorie", "kategorie_vypoctena", "trida_U"] df_vyber_agregovany = ( df_vyber.groupby(GRP) .agg( Pocet_v_Vyberu=("IdHlasenky", "count"), Suma_Vah_Kombinace=("combined_weight", "sum"), ) .reset_index() ) df_suma_hlavni = ( df_vyber.groupby("ENEX_kategorie")["combined_weight"] .sum() .reset_index() .rename(columns={"combined_weight": "Suma_Vah_Hlavni"}) ) df_klic = pd.merge(df_vyber_agregovany, df_suma_hlavni, on="ENEX_kategorie", how="left") df_klic["Procentualni_Podil"] = np.where( df_klic["Suma_Vah_Hlavni"] > 0, df_klic["Suma_Vah_Kombinace"] / df_klic["Suma_Vah_Hlavni"], 0, ) # -------------------------------------------------- # 4. PŘENOS KLÍČE A CÍLOVÉ MNOŽSTVÍ V POPULACI # -------------------------------------------------- print("Aplikuji výběrový klíč na reálné počty z DB...") df_klic = pd.merge(df_klic, df_populace_counts, on="ENEX_kategorie", how="left") df_klic["Target_Populace_Kombinace"] = ( df_klic["Pocet_v_Populaci_Real"].fillna(0) * df_klic["Procentualni_Podil"] ) # -------------------------------------------------- # 5. VEKTORIZOVANÝ EXPANZNÍ GENERÁTOR (fyzická replikace řádků) # -------------------------------------------------- print("\nSpouštím expanzní generátor populace (proporční dle combined_weight)...") df_vyber_s_klice = pd.merge( df_vyber, df_klic[GRP + ["Target_Populace_Kombinace"]], on=GRP, how="left", ) df_vyber_s_klice["Target_Populace_Kombinace"] = df_vyber_s_klice["Target_Populace_Kombinace"].fillna(0.0) suma_vah_podskupina = df_vyber_s_klice.groupby(GRP)["combined_weight"].transform("sum") maska_vahy_platna = df_vyber_s_klice["combined_weight"].notna() df_vyber_s_klice["podil_v_podskupine"] = np.where( maska_vahy_platna & (suma_vah_podskupina > 0), df_vyber_s_klice["combined_weight"] / suma_vah_podskupina, 0.0, ) df_vyber_s_klice["target_na_jeden_radek"] = ( df_vyber_s_klice["Target_Populace_Kombinace"] * df_vyber_s_klice["podil_v_podskupine"] ).fillna(0.0) df_vyber_s_klice["cela_cast"] = np.floor(df_vyber_s_klice["target_na_jeden_radek"]).astype(int) df_vyber_s_klice["desetinny_zbytek"] = ( df_vyber_s_klice["target_na_jeden_radek"] - df_vyber_s_klice["cela_cast"] ) print("Generuji celé budovy...") df_cele_budovy = df_vyber_s_klice.loc[ df_vyber_s_klice.index.repeat(df_vyber_s_klice["cela_cast"]) ].copy() df_cele_budovy["vaha2"] = 1.0 df_cele_budovy["typ_radku_populace"] = "Cela_budova" print("Generuji umělé zbytky budov...") df_zbytky = df_vyber_s_klice[df_vyber_s_klice["desetinny_zbytek"] > 0.00001].copy() df_zbytky["vaha2"] = df_zbytky["desetinny_zbytek"] df_zbytky["typ_radku_populace"] = "Umely_zbytek_budovy" print("Slucuji generovaná data do výsledné tabulky...") df_populace_celkova = pd.concat([df_cele_budovy, df_zbytky], ignore_index=True) sloupce_k_odstraneni = [ "Target_Populace_Kombinace", "podil_v_podskupine", "target_na_jeden_radek", "cela_cast", "desetinny_zbytek", ] df_populace_celkova = df_populace_celkova.drop(columns=sloupce_k_odstraneni, errors="ignore") for col in ["celk_dod_ener", "celkprim", "PlochaEvzB", "combined_weight", "combined_weight_norm"]: if col in df_populace_celkova.columns: df_populace_celkova[col] = pd.to_numeric( df_populace_celkova[col].astype(str).str.replace(",", ".", regex=False), errors="coerce" ) # -------------------------------------------------- # 6. KONTROLA VALIDITY GENERÁTORU A VÁŽENÝCH PRŮMĚRŮ # -------------------------------------------------- vysledna_suma_vaha2 = df_populace_celkova["vaha2"].sum() rozdil_oproti_db = vysledna_suma_vaha2 - celkem_populace_db def _vaz_prumer(frame, hodnota, vahy): """Vážený průměr hodnoty; vahy je Series stejné délky jako frame.""" x = frame[hodnota] m = x.notna() & vahy.notna() & (vahy > 0) return (x[m] * vahy[m]).sum() / vahy[m].sum() if m.any() else np.nan # A. celk_dod_ener – výběr (combined_weight vs combined_weight_norm) vs populace (vaha2) vaz_prumer_vyber = _vaz_prumer(df_vyber, "celk_dod_ener", df_vyber["combined_weight"]) vaz_prumer_vyber_norm = _vaz_prumer(df_vyber, "celk_dod_ener", df_vyber["combined_weight_norm"]) vaz_prumer_populace = _vaz_prumer(df_populace_celkova, "celk_dod_ener", df_populace_celkova["vaha2"]) # B. celkprim vážené PLOCHOU – výběr (orig váha vs normovaná váha) vs populace (vaha2*plocha) vaz_prumer_prim_vyber = _vaz_prumer( df_vyber, "celkprim", df_vyber["combined_weight"] * df_vyber["PlochaEvzB"] ) vaz_prumer_prim_vyber_norm = _vaz_prumer( df_vyber, "celkprim", df_vyber["combined_weight_norm"] * df_vyber["PlochaEvzB"] ) vaz_prumer_prim_populace = _vaz_prumer( df_populace_celkova, "celkprim", df_populace_celkova["vaha2"] * df_populace_celkova["PlochaEvzB"] ) print("\n" + "=" * 80) print("KONTROLA GENERÁTORU CELÉ POPULACE:") print(f" - Počet řádků ve výběru s chybějící vahou: {pocet_chybejici_vaha:,}") print(f" - Celkový počet fyzických řádků populace: {len(df_populace_celkova):,}") print(f" * Celých budov (vaha2=1.0): " f"{(df_populace_celkova['typ_radku_populace']=='Cela_budova').sum():,}") print(f" * Umělých řádků (zbytků): " f"{(df_populace_celkova['typ_radku_populace']=='Umely_zbytek_budovy').sum():,}") print(f" - Suma 'vaha2': {vysledna_suma_vaha2:,.3f}") print(f" - Reálná velikost populace v DuckDB: {celkem_populace_db:,}") print(f" - Odchylka: {rozdil_oproti_db:,.6f}") print("-" * 80) print("KONTROLA celk_dod_ener:") print(f" - Vážený průměr VÝBĚR (combined_weight): {vaz_prumer_vyber:,.4f}") print(f" - Vážený průměr VÝBĚR (combined_weight_norm):{vaz_prumer_vyber_norm:,.4f}") print(f" - Vážený průměr POPULACE (vaha2): {vaz_prumer_populace:,.4f}") print("-" * 80) print("KONTROLA celkprim VÁŽENÉ PLOCHOU:") print(f" - Vážený průměr VÝBĚR (weight*plocha): {vaz_prumer_prim_vyber:,.4f}") print(f" - Vážený průměr VÝBĚR (weight_norm*plocha): {vaz_prumer_prim_vyber_norm:,.4f}") print(f" - Vážený průměr POPULACE (vaha2*plocha): {vaz_prumer_prim_populace:,.4f}") rozdil_prim = abs(vaz_prumer_prim_vyber - vaz_prumer_prim_populace) rozdil_prim_norm = abs(vaz_prumer_prim_vyber_norm - vaz_prumer_prim_populace) print(f" Rozdíl (orig vs populace): {rozdil_prim:,.4f}") print(f" Rozdíl (norm vs populace): {rozdil_prim_norm:,.4f}") print("-" * 80) # -------------------------------------------------- # 6a. STEJNÉ KONTROLY ZVLÁŠŤ PRO RD A ZVLÁŠŤ PRO BD # -------------------------------------------------- if "skupina_rdbd" in df_vyber.columns and "skupina_rdbd" in df_populace_celkova.columns: print("KONTROLA ZVLÁŠŤ PODLE SKUPINY (RD / BD):") for skupina in ["RD", "BD"]: sub_vyber = df_vyber[df_vyber["skupina_rdbd"] == skupina] sub_populace = df_populace_celkova[df_populace_celkova["skupina_rdbd"] == skupina] vp_dod_vyber = _vaz_prumer(sub_vyber, "celk_dod_ener", sub_vyber["combined_weight"]) vp_dod_vyber_norm = _vaz_prumer(sub_vyber, "celk_dod_ener", sub_vyber["combined_weight_norm"]) vp_dod_populace = _vaz_prumer(sub_populace, "celk_dod_ener", sub_populace["vaha2"]) vp_prim_vyber = _vaz_prumer( sub_vyber, "celkprim", sub_vyber["combined_weight"] * sub_vyber["PlochaEvzB"] ) vp_prim_vyber_norm = _vaz_prumer( sub_vyber, "celkprim", sub_vyber["combined_weight_norm"] * sub_vyber["PlochaEvzB"] ) vp_prim_populace = _vaz_prumer( sub_populace, "celkprim", sub_populace["vaha2"] * sub_populace["PlochaEvzB"] ) print(f" --- {skupina} (výběr n={len(sub_vyber):,}, populace n={len(sub_populace):,}, " f"suma vaha2={sub_populace['vaha2'].sum():,.1f}) ---") print(f" celk_dod_ener - výběr (orig): {vp_dod_vyber:,.4f} | výběr (norm): {vp_dod_vyber_norm:,.4f} | populace: {vp_dod_populace:,.4f}") print(f" celkprim (plochou) - výběr (orig): {vp_prim_vyber:,.4f} | výběr (norm): {vp_prim_vyber_norm:,.4f} | populace: {vp_prim_populace:,.4f}") print("-" * 80) # -------------------------------------------------- # 6b. SUMA CELKOVÉ PODLAHOVÉ PLOCHY (PlochaEvzB) # -------------------------------------------------- suma_plocha_populace = (df_populace_celkova["PlochaEvzB"] * df_populace_celkova["vaha2"]).sum() suma_plocha_vyber_nevazene = df_vyber["PlochaEvzB"].sum() print("KONTROLA CELKOVÉ PLOCHY (PlochaEvzB):") print(f" - Suma PlochaEvzB v populaci (váženo vaha2): {suma_plocha_populace:,.1f} m²") print(f" - Suma PlochaEvzB ve výběru (nevážená, info): {suma_plocha_vyber_nevazene:,.1f} m²") if "skupina_rdbd" in df_populace_celkova.columns: for skupina in ["RD", "BD"]: sub_populace = df_populace_celkova[df_populace_celkova["skupina_rdbd"] == skupina] suma_plocha_skupina = (sub_populace["PlochaEvzB"] * sub_populace["vaha2"]).sum() print(f" * {skupina}: {suma_plocha_skupina:,.1f} m²") print("=" * 80) # -------------------------------------------------- # 7. ULOŽENÍ VÝSLEDNÉ POPULACE # -------------------------------------------------- output_populace_csv = "ENEX_populace_rozsirena_rezidence.csv" print(f"\nUkládám kompletní populaci s 'vaha2' do {output_populace_csv}...") df_populace_celkova.to_csv( output_populace_csv, sep=";", decimal=",", encoding="utf-8-sig", index=False, ) print("Hotovo! Dataset fyzicky rozšířen na celou populaci (RD+BD).")
